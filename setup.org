* Package Infrastructure & Safe Bootstrap
#+BEGIN_SRC emacs-lisp
  (require 'package)

  (setq package-archives
        '(("gnu"   . "https://elpa.gnu.org/packages/")
          ("melpa" . "https://melpa.org/packages/")))

  (package-initialize)

  ;; Install a package only when it is actually missing.
  ;; This avoids refreshing package archives on every startup.
  (defun my/ensure-package (package)
    "Install PACKAGE if it is not already installed."
    (unless (package-installed-p package)
      (unless package-archive-contents
        (package-refresh-contents))
      (condition-case err
          (package-install package)
        (error
         (display-warning
          'my-package-bootstrap
          (format "Could not install %s: %s"
                  package
                  (error-message-string err))
          :warning)))))

  ;; use-package is the only package that must exist before the
  ;; rest of this configuration can be evaluated.
  (my/ensure-package 'use-package)
  (require 'use-package)

  ;; Packages declared below may be installed automatically when
  ;; their use-package declaration is evaluated, but most of them
  ;; are deferred so they do not load during startup.
  (setq use-package-always-ensure t)
#+END_SRC

(defun open-line-above ()
  "Open a new line above the current line."
  (interactive)
  (beginning-of-line)
  (newline)
  (forward-line -1)
  (indent-for-tab-command))

(defun open-line-below ()
  "Open a new line below the current line."
  (interactive)
  (end-of-line)
  (newline)
  (indent-for-tab-command))

(defun duplicate-line ()
  "Duplicate the current line."
  (interactive)
  (beginning-of-line)
  (set-mark-command nil)
  (end-of-line)
  (kill-ring-save (region-beginning) (region-end))
  (newline)
  (yank))

(defun my-org-screenshot ()
  "Take a screenshot into a timestamped file in the Org buffer directory."
  (interactive)
  (let ((filename
         (concat
          (make-temp-name
           (concat
            (file-name-nondirectory (buffer-file-name))
            "_imgs/"
            (format-time-string "%Y%m%d_%H%M%S_")))
          ".png")))
    (unless (file-exists-p (file-name-directory filename))
      (make-directory (file-name-directory filename) t))
    (pcase system-type
      ('darwin
       (shell-command (concat "pngpaste " filename)))
      ('gnu/linux
       (call-process "import" nil nil nil filename)))
    (when (file-exists-p filename)
      (insert (concat "[[file:" filename "]]"))
      (org-display-inline-images))))

(defun my-insert-file-name (filename &optional args)
  "Insert FILENAME.

With prefix argument `-', insert a relative path.
With another prefix argument, insert an absolute path."
  (interactive "*fInsert file name: \nP")
  (cond
   ((eq '- args)
    (insert (file-relative-name filename)))
   ((not (null args))
    (insert (expand-file-name filename)))
   (t
    (insert filename))))

(use-package flycheck
  :ensure t
  :defer nil
  :config
  ;; Flycheck ships `javascript-oxlint', which consumes oxlint's checkstyle
  ;; XML.  The previous hand-rolled `--format=unix' parser only understood
  ;; `[Warning|Error/rule]' lines, so any oxlint exit that printed something
  ;; else (config/type-aware errors, crashes) surfaced as
  ;; "cannot read oxlint's output".  Prefer the built-in checker.
  (setq flycheck-checkers
        (cons 'javascript-oxlint
              (delq 'javascript-oxlint flycheck-checkers)))
  ;; oxlint also lints the <script> frontmatter of .astro files, so teach
  ;; the checker about `astro-ts-mode'.
  (setf (flycheck-checker-get 'javascript-oxlint 'modes)
        (cons 'astro-ts-mode
              (flycheck-checker-get 'javascript-oxlint 'modes)))
  (dolist (hook '(js-mode-hook js-ts-mode-hook
                  typescript-ts-mode-hook tsx-ts-mode-hook
                  astro-ts-mode-hook))
    (add-hook hook #'flycheck-mode)))

* Agent Shell (ACP agents: Claude, Codex, Gemini, Pi, ...)
#+BEGIN_SRC emacs-lisp
;; agent-shell talks to any ACP agent.  Install the agent CLI first, e.g.:
;;   npm install -g pi-acp            ; this machine already has it
;;   npm install -g @agentclientprotocol/claude-agent-acp
;; `pi-acp' and `gemini' are on PATH here, so no extra setup is needed.
(use-package agent-shell
  :ensure t
  :defer t
  :bind (("C-c a" . agent-shell)          ; start/reuse a shell
         ("C-c A" . agent-shell-prompt-compose))
  :init
  ;; Default agent for `M-x agent-shell'.  Identifiers live in
  ;; `agent-shell-agent-configs' (e.g. 'pi, 'gemini-cli).  Set at startup
  ;; (not :custom, which would be deferred along with the package).
  (setq agent-shell-preferred-agent-config 'pi)
  :config
  ;; Keep transcripts/screenshots out of project trees (~/.emacs.d/agent-shell/).
  ;; Comment this out if you prefer the default .agent-shell/ in each project.
  (defun my/agent-shell-dot-subdir (subdir)
    "Return path to store agent-shell SUBDIR under `user-emacs-directory'."
    (let* ((cwd (string-remove-suffix "/" (agent-shell-cwd)))
           (sanitized (replace-regexp-in-string
                       "/" "-" (string-remove-prefix "/" cwd))))
      (expand-file-name subdir
                        (locate-user-emacs-file
                         (concat "agent-shell/" sanitized)))))
  (setopt agent-shell-dot-subdir-function #'my/agent-shell-dot-subdir))
#+END_SRC

;; astro-ls (Volar-based) requires an explicit TypeScript SDK path and
;; rejects a workspace that only has TypeScript 7 (which no longer ships
;; typescript.js).  Point it at the nearest node_modules/typescript/lib
;; that still contains typescript.js.
(defun my/find-typescript-tsdk (start)
  "Return nearest `node_modules/typescript/lib' under START with typescript.js."
  (let ((dir (file-name-as-directory (expand-file-name start))) found)
    (while (and (not found)
                (not (equal dir (file-name-directory (directory-file-name dir)))))
      (let ((cand (expand-file-name "node_modules/typescript/lib" dir)))
        (when (file-exists-p (expand-file-name "typescript.js" cand))
          (setq found cand)))
      (setq dir (file-name-directory (directory-file-name dir))))
    found))

(defun my/astro-ls-contact (_interactive project)
  "Eglot contact for astro-ls, injecting the project's TypeScript SDK."
  (list "astro-ls" "--stdio"
        :initializationOptions
        (list :typescript
              (list :tsdk (or (my/find-typescript-tsdk default-directory)
                              (expand-file-name "node_modules/typescript/lib"
                                                (project-root project)))))))

;; Eglot is built into modern Emacs versions, so do not install a
;; second copy from ELPA/MELPA.
(use-package eglot
  :ensure nil
  :defer nil
  :hook
  ((typescript-ts-mode . eglot-ensure)
   (tsx-ts-mode . eglot-ensure)
   (js-ts-mode . eglot-ensure)
   (js-mode . eglot-ensure)
   (astro-ts-mode . eglot-ensure))
  :config
  ;; eglot-typescript-preset (installed via package-selected-packages) adds
  ;; its own eglot-server-programs entry at after-init, defaulting to
  ;; `typescript-language-server'.  Since it is added after this config it
  ;; shadows the tsgo entry below and would fail on workspaces using
  ;; TypeScript 7 (no tsserver.js).  Disable its auto-setup.
  (setq eglot-typescript-preset-auto-setup nil)
  ;; Use the TypeScript 7 native server (tsgo) directly.  Going through
  ;; rassumfrassum as a multiplexer kept aborting the whole session when a
  ;; server exited, so rass is not used at all.
  (add-to-list 'eglot-server-programs
               '(((js-mode :language-id "javascript")
                  (js-ts-mode :language-id "javascript")
                  (typescript-ts-mode :language-id "typescript")
                  (tsx-ts-mode :language-id "typescriptreact"))
                 . ("tsgo" "--lsp" "--stdio")))
  ;; Astro language server (astro-ls).  Install globally with
  ;;   bun add -g @astrojs/language-server
  (add-to-list 'eglot-server-programs
               (cons 'astro-ts-mode #'my/astro-ls-contact)))

* Format on Save (oxfmt)
#+BEGIN_SRC emacs-lisp
;; Run oxfmt directly on save for JS/TS files.  This keeps formatting out
;; of the LSP multiplexer, which is fragile (see the Eglot preset above).
(defun my/oxfmt-buffer ()
  "Format the current buffer with oxfmt, leaving it unchanged on failure."
  (when (and buffer-file-name
             (executable-find "oxfmt")
             (derived-mode-p 'js-mode 'js-ts-mode
                             'tsx-ts-mode 'typescript-ts-mode))
    (let ((out (get-buffer-create " *oxfmt*")))
      (with-current-buffer out (erase-buffer))
      (let ((status (call-process-region
                     (point-min) (point-max) "oxfmt" nil out nil
                     "--stdin-filepath" buffer-file-name)))
        (when (and (integerp status) (zerop status))
          (let ((pt (point)))
            (erase-buffer)
            (insert-buffer-substring out)
            (goto-char (min pt (point-max))))))
      (kill-buffer out))))

(add-hook 'before-save-hook #'my/oxfmt-buffer)
#+END_SRC
